"""Semantic search over knowledge-base articles using Valkey's vector index.

Usage: python3 /app/search.py < query.json
where query.json contains a single query embedding (JSON list of 64 floats).
Prints the ids of the 5 articles most cosine-similar to the query, one per
line, most similar first.

All similarity computation happens inside Valkey (FT.SEARCH KNN over the
`idx:articles` vector index); this script never reads the source data file
and never computes similarities itself.
"""
import json
import struct
import sys

import valkey

INDEX_NAME = "idx:articles"
KEY_PREFIX = "kb:"
DIM = 64
K = 5


def main() -> None:
    query = json.load(sys.stdin)
    if not isinstance(query, list) or len(query) != DIM:
        sys.exit(f"query embedding must be a JSON list of {DIM} floats")

    query_blob = struct.pack(f"<{DIM}f", *[float(x) for x in query])

    c = valkey.Valkey(host="127.0.0.1", port=6379, decode_responses=False)

    # Exact KNN search executed by Valkey's search module.  The KNN clause
    # orders the reply by ascending distance (cosine distance = 1 - cosine
    # similarity), so the first document is the most similar article.
    # RETURN 1 id fetches the article id stored in each hash.
    reply = c.execute_command(
        "FT.SEARCH", INDEX_NAME,
        f"*=>[KNN {K} @embedding $vec AS score]",
        "RETURN", "1", "id",
        "PARAMS", "2", "vec", query_blob,
        "DIALECT", "2",
    )

    if not isinstance(reply, list) or len(reply) < 1 + 2 * K:
        sys.exit(f"unexpected FT.SEARCH reply: {reply!r}")

    # Reply: [total, key, [field, value], key, [field, value], ...]
    results = []
    for key, attrs in zip(reply[1::2], reply[2::2]):
        key = key.decode() if isinstance(key, bytes) else str(key)
        article_id = None
        if isinstance(attrs, (list, tuple)):
            for i in range(0, len(attrs) - 1, 2):
                field = attrs[i].decode() if isinstance(attrs[i], bytes) else str(attrs[i])
                if field == "id":
                    val = attrs[i + 1]
                    article_id = val.decode() if isinstance(val, bytes) else str(val)
        results.append(article_id if article_id else key[len(KEY_PREFIX):])

    for article_id in results:
        print(article_id)


if __name__ == "__main__":
    main()
