"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service
"""
import json
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"
_CACHE_TTL = 300  # seconds
_cache = valkey.Valkey(host="127.0.0.1", port=6379, socket_connect_timeout=3, socket_timeout=3)


def _keys(product_id: int) -> tuple[str, str]:
    product_id = int(product_id)
    return f"catalog:product:{product_id}", f"catalog:product-lock:{product_id}"


def get_product(product_id: int) -> dict:
    product_id = int(product_id)
    key, lock_key = _keys(product_id)
    # Serialize fills with price changes across processes. In particular, a
    # slow inventory GET must not refill the cache after a successful PUT.
    with _cache.lock(lock_key, timeout=30, blocking_timeout=30):
        cached = _cache.get(key)
        if cached is not None:
            return json.loads(cached)

        with urllib.request.urlopen(f"{INVENTORY_URL}/products/{product_id}", timeout=10) as resp:
            product = json.load(resp)
        _cache.set(key, json.dumps(product), ex=_CACHE_TTL)
        return product


def update_price(product_id: int, price: float) -> None:
    product_id = int(product_id)
    key, lock_key = _keys(product_id)
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{product_id}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with _cache.lock(lock_key, timeout=30, blocking_timeout=30):
        with urllib.request.urlopen(req, timeout=10) as resp:
            resp.read()
        # Only invalidate after the inventory service has accepted the change.
        _cache.delete(key)
