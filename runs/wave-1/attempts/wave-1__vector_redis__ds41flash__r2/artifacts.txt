#!/usr/bin/env python3
"""Semantic search over the knowledge-base articles stored in Redis.

Usage:  python3 /app/search.py < query.json

Reads a single query embedding (JSON list of 64 floats) from stdin at most once,
then asks Redis (via the vector-set VSIM command) for the 5 most similar
articles by cosine similarity and prints their ids, most similar first.

All stored vectors and similarity computation live inside Redis; this script
never reads /data/articles.jsonl and never computes similarities itself.
"""
import json
import sys

import redis

VSET_KEY = "articles:vset"
TOP_K = 5


def read_query_vector():
    data = sys.stdin.read().strip()
    if not data:
        sys.exit("no query embedding on stdin")
    vec = json.loads(data)
    # Tolerate either a bare list or a single-element list of lists.
    if vec and isinstance(vec[0], list):
        vec = vec[0]
    if not isinstance(vec, list) or not vec:
        sys.exit("query embedding must be a non-empty JSON list of floats")
    return [float(x) for x in vec]


def main():
    vector = read_query_vector()

    r = redis.Redis(host="127.0.0.1", port=6379, decode_responses=True)

    # VSIM runs the similarity search inside Redis; default metric is cosine.
    result = r.execute_command(
        "VSIM", VSET_KEY, "VALUES", len(vector),
        *[repr(x) for x in vector],
        "COUNT", TOP_K,
    )

    for elem in result[:TOP_K]:
        print(elem)


if __name__ == "__main__":
    main()
