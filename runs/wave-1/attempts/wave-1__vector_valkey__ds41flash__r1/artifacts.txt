#!/usr/bin/env python3
"""Semantic search over the knowledge-base articles stored in Valkey.

Usage:  python3 /app/search.py < query.json

`query.json` is a JSON list of 64 floats (the query embedding). The script
prints the ids of the 5 articles most similar to the query, one per line, most
similar first.

All similarity computation happens *inside* Valkey: this script only sends the
query vector to the server and runs an FT.SEARCH KNN query against the
`articles` vector index (which was built by /app/load_articles.py). It never
reads /data/articles.jsonl and never computes cosine similarities itself.
"""
import json
import struct
import sys

import valkey

INDEX_NAME = "articles"
VECTOR_FIELD = "embedding"
DIM = 64
TOP_K = 5


def main():
    query = json.loads(sys.stdin.read())
    if not isinstance(query, list) or len(query) != DIM:
        raise SystemExit(
            "query.json must be a JSON list of %d floats" % DIM
        )
    query_blob = struct.pack("<%df" % DIM, *[float(x) for x in query])

    client = valkey.Valkey(host="127.0.0.1", port=6379, decode_responses=True)

    # KNN search executed server-side by the valkey-search module; results come
    # back already sorted by (cosine) distance, closest first.
    reply = client.execute_command(
        "FT.SEARCH", INDEX_NAME,
        "*=>[KNN %d @%s $vec]" % (TOP_K, VECTOR_FIELD),
        "PARAMS", "2", "vec", query_blob,
        "RETURN", "1", "id",
        "DIALECT", "2",
    )

    # reply = [count, key1, [field, value, ...], key2, [field, value, ...], ...]
    docs = reply[1:]
    for i in range(0, len(docs), 2):
        fields = docs[i + 1]
        fmap = dict(zip(fields[0::2], fields[1::2]))
        print(fmap["id"])


if __name__ == "__main__":
    main()
