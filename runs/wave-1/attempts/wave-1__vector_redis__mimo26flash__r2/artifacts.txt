#!/usr/bin/env python3
"""Semantic search over 1,000 KB articles stored in Redis.

Reads a query embedding (JSON list of 64 floats) from stdin and prints the ids
of the 5 most similar articles (cosine similarity, most similar first).

The similarity search runs inside Redis via a RediSearch KNN vector query;
this script never reads the source data file nor computes any similarity.
"""
import json
import struct
import sys

import redis

REDIS_HOST = "127.0.0.1"
REDIS_PORT = 6379
INDEX = "articles_idx"
DIM = 64
TOP_K = 5


def main() -> None:
    query = json.load(sys.stdin)
    if not isinstance(query, list) or len(query) != DIM:
        raise SystemExit(f"expected a JSON list of {DIM} floats on stdin")
    # pack to little-endian float32 bytes for the Redis vector param
    vec = struct.pack(f"<{DIM}f", *(float(x) for x in query))

    r = redis.Redis(host=REDIS_HOST, port=REDIS_PORT, decode_responses=False)
    res = r.execute_command(
        "FT.SEARCH", INDEX,
        f"*=>[KNN {TOP_K} @embedding $vec AS dist]",
        "PARAMS", "2", "vec", vec,
        "SORTBY", "dist",
        "LIMIT", "0", str(TOP_K),
        "DIALECT", "2",
        "RETURN", "1", "id",
    )

    for item in res.get(b"results", []):
        article_id = item[b"extra_attributes"][b"id"]
        sys.stdout.buffer.write(article_id + b"\n")


if __name__ == "__main__":
    main()
