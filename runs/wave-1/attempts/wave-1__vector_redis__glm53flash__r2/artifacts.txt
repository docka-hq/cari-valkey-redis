#!/usr/bin/env python3
"""Semantic search over the knowledge-base articles stored in Redis.

Usage:  python3 /app/search.py < query.json

query.json holds one query embedding: a JSON list of 64 floats.
Prints the ids of the 5 most similar articles (cosine similarity),
one per line, most similar first.

All similarity math runs inside Redis: the articles' embeddings live in
Redis hashes and the KNN search is executed by the RediSearch FLAT
vector index (exact search, COSINE distance).  This script only forwards
the query vector and prints the ids Redis returns.
"""
import json
import struct
import sys

import redis

HOST = "127.0.0.1"
PORT = 6379
INDEX_NAME = "articles:idx"
DIM = 64
K = 5


def main() -> None:
    query = json.load(sys.stdin)
    if isinstance(query, dict):  # tolerate {"embedding": [...]}
        query = query.get("embedding")
    if not isinstance(query, list) or len(query) != DIM:
        raise SystemExit(f"query must be a JSON list of {DIM} floats")

    qblob = struct.pack(f"<{DIM}d", *query)  # native little-endian doubles

    r = redis.Redis(host=HOST, port=PORT)
    try:
        res = r.execute_command(
            "FT.SEARCH", INDEX_NAME,
            f"*=>[KNN {K} @embedding $q]",
            "RETURN", 1, "id",
            "SORTBY", "__embedding_score", "ASC",  # ascending cosine distance
            "LIMIT", 0, K,
            "PARAMS", 2, "q", qblob,
            "DIALECT", 2,
        )
    except redis.exceptions.ResponseError as e:
        raise SystemExit(f"Redis vector search failed: {e}")

    rows = res[b"results"]
    if len(rows) < K:
        raise SystemExit(f"index {INDEX_NAME!r} returned only {len(rows)} of {K} results")

    for row in rows[:K]:
        print(row[b"extra_attributes"][b"id"].decode())


if __name__ == "__main__":
    main()
