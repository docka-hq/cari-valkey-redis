"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Reads of a product are cached in Valkey so that the (slow) inventory service is
only hit on a cache miss.  The cache is shared by every storefront process,
entries live for CACHE_TTL seconds, and a price change invalidates the entry so
the next read in *any* process sees the new price.
"""
import json
import os
import urllib.request

INVENTORY_URL = "http://127.0.0.1:8000"

VALKEY_HOST = os.environ.get("VALKEY_HOST", "127.0.0.1")
VALKEY_PORT = int(os.environ.get("VALKEY_PORT", "6379"))
VALKEY_DB = int(os.environ.get("VALKEY_DB", "0"))

CACHE_TTL = 300  # seconds; cached products expire after 5 minutes

_CACHE_PREFIX = "catalog:product"
_VERSION_PREFIX = "catalog:version"


def _make_client():
    """Return a Valkey client, falling back to the redis client if needed."""
    try:
        import valkey  # type: ignore

        return valkey.Valkey(
            host=VALKEY_HOST,
            port=VALKEY_PORT,
            db=VALKEY_DB,
            decode_responses=True,
        )
    except ImportError:
        import redis  # type: ignore

        return redis.Redis(
            host=VALKEY_HOST,
            port=VALKEY_PORT,
            db=VALKEY_DB,
            decode_responses=True,
        )


# Connection pool is shared by all threads/processes; commands are lazy.
_client = _make_client()


def _cache_key(product_id: int) -> str:
    return f"{_CACHE_PREFIX}:{product_id}"


def _version_key(product_id: int) -> str:
    return f"{_VERSION_PREFIX}:{product_id}"


# Store the product only if no price update happened while we were fetching it.
# This keeps a slow in-flight read from writing stale data back over a value
# that a concurrent update_price() has just invalidated.
_SET_IF_UNCHANGED = """
local current = redis.call('GET', KEYS[2]) or '0'
if current ~= ARGV[1] then
  return 0
end
redis.call('SET', KEYS[1], ARGV[2], 'EX', ARGV[3])
return 1
"""


def _fetch_from_inventory(product_id: int) -> dict:
    with urllib.request.urlopen(
        f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10
    ) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    product_id = int(product_id)
    cache_key = _cache_key(product_id)
    version_key = _version_key(product_id)

    # Fast path: a recently read product is served straight from Valkey.
    try:
        cached = _client.get(cache_key)
        if cached is not None:
            return json.loads(cached)
        version = _client.get(version_key) or "0"
    except Exception:  # noqa: BLE001 - cache outage must not break the storefront
        return _fetch_from_inventory(product_id)

    product = _fetch_from_inventory(product_id)

    # Cache the product for CACHE_TTL, but do not clobber a newer value if an
    # update_price() landed while we were talking to the inventory service.
    try:
        _client.eval(
            _SET_IF_UNCHANGED,
            2,
            cache_key,
            version_key,
            version,
            json.dumps(product),
            CACHE_TTL,
        )
    except Exception:  # noqa: BLE001
        pass

    return product


def update_price(product_id: int, price: float) -> None:
    product_id = int(product_id)
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate the shared cache so the next get_product() in any process
    # re-reads the (now updated) price from the inventory service.  Bumping the
    # version first stops an in-flight read from re-populating stale data.
    try:
        _client.incr(_version_key(product_id))
        _client.delete(_cache_key(product_id))
    except Exception:  # noqa: BLE001
        pass
