#!/usr/bin/env python3
"""Semantic search over the knowledge-base articles stored in Redis.

Usage:
    python3 /app/search.py < query.json

`query.json` holds a single query embedding: a JSON list of 64 floats.

The script prints the ids of the 5 most similar articles (cosine
similarity), one per line, most similar first.

All articles and the similarity computation live inside Redis: the
embeddings are stored in the `articles:vectorset` vector set and the
ranking is delegated to the Redis `VSIM` command.  This script never
reads /data/articles.jsonl and never computes similarities itself.
"""
import json
import os
import sys

import redis

VECTOR_SET_KEY = "articles:vectorset"
TOP_K = 5
HOST = os.environ.get("REDIS_HOST", "127.0.0.1")
PORT = int(os.environ.get("REDIS_PORT", "6379"))


def parse_query(raw):
    """Return the query embedding from the raw stdin text.

    Accepts a bare JSON list of floats as well as a JSON object that wraps
    the embedding under a common field name.
    """
    data = json.loads(raw)
    if isinstance(data, dict):
        for field in ("embedding", "vector", "query", "query_embedding"):
            if field in data:
                data = data[field]
                break
    # Tolerate a single-element list wrapping the vector.
    if isinstance(data, list) and len(data) == 1 and isinstance(data[0], list):
        data = data[0]
    if not isinstance(data, list):
        raise ValueError("query must be a JSON list of floats")
    return [float(x) for x in data]


def main():
    raw = sys.stdin.read()
    query = parse_query(raw)
    if not query:
        return

    client = redis.Redis(host=HOST, port=PORT, decode_responses=True)

    # VSIM runs the similarity search (HNSW / exact) inside Redis.
    # TRUTH forces an exact result, which is cheap for this dataset size.
    reply = client.execute_command(
        "VSIM", VECTOR_SET_KEY, "VALUES", len(query), *query,
        "COUNT", TOP_K, "TRUTH",
    )

    # The reply is a list of element names (optionally [name, score] pairs).
    idents = [
        item[0] if isinstance(item, (list, tuple)) else item
        for item in reply
    ]

    sys.stdout.write("\n".join(str(i) for i in idents) + "\n")


if __name__ == "__main__":
    main()
