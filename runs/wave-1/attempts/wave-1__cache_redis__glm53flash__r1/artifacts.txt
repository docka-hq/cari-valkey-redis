"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Reads are cached in Redis (shared by every storefront process); cached
products expire after CACHE_TTL_SECONDS.
"""
import json
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"

_REDIS_HOST = "127.0.0.1"
_REDIS_PORT = 6379

# One module-level connection pool shared by all users of this process; it is
# process-local only (Redis itself is what makes the cache shared).
_pool = redis.ConnectionPool(
    host=_REDIS_HOST,
    port=_REDIS_PORT,
    db=0,
    decode_responses=True,
    socket_timeout=2,
    socket_connect_timeout=2,
    health_check_interval=30,
)
_r = redis.Redis(connection_pool=_pool)

# How long a cached product stays valid.
CACHE_TTL_SECONDS = 300


def _key(product_id: int) -> str:
    """Redis key under which a product is cached."""
    return f"product:{int(product_id)}"


def get_product(product_id: int) -> dict:
    pid = int(product_id)
    key = _key(pid)

    try:
        cached = _r.get(key)
    except redis.RedisError:
        cached = None
    if cached is not None:
        return json.loads(cached)

    # Cache miss: fetch from the inventory service, then cache it.
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{pid}", timeout=10) as resp:
        product = json.load(resp)

    try:
        _r.set(key, json.dumps(product), ex=CACHE_TTL_SECONDS)
    except redis.RedisError:
        pass  # serving uncached data is better than failing on a cache error
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # The inventory service has the new price, so any cached copy (in this or
    # any other process) is stale: drop it so the next read refetches.
    try:
        _r.delete(_key(product_id))
    except redis.RedisError:
        pass
