"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service
"""
import json
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"
CACHE_TTL_SECONDS = 300
_cache = redis.Redis(host="127.0.0.1", port=6379, decode_responses=True)


def _product_key(product_id: int) -> str:
    return f"catalog:product:v1:{int(product_id)}"


def get_product(product_id: int) -> dict:
    key = _product_key(product_id)
    cached = _cache.get(key)
    if cached is not None:
        return json.loads(cached)

    # Serialize misses with writes across processes: a read started before a
    # price change must not populate the cache after that change is committed.
    with _cache.lock(f"{key}:lock", timeout=30):
        cached = _cache.get(key)
        if cached is not None:
            return json.loads(cached)
        with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
            product = json.load(resp)
        _cache.setex(key, CACHE_TTL_SECONDS, json.dumps(product))
        return product


def update_price(product_id: int, price: float) -> None:
    key = _product_key(product_id)
    with _cache.lock(f"{key}:lock", timeout=30):
        body = json.dumps({"price": price}).encode()
        req = urllib.request.Request(
            f"{INVENTORY_URL}/products/{int(product_id)}",
            data=body,
            method="PUT",
            headers={"Content-Type": "application/json"},
        )
        with urllib.request.urlopen(req, timeout=10) as resp:
            resp.read()
        # Only invalidate after the inventory service has accepted the update.
        _cache.delete(key)
