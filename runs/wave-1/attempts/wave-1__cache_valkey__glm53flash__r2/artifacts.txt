"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Reads go through a shared cache in Valkey (127.0.0.1:6379), so products read
recently by ANY storefront process are served from Valkey instead of the slow
inventory service. Each product is stored under the key "catalog:product:<id>"
as its JSON representation and expires after CACHE_TTL_SECONDS (5 minutes).

update_price() writes through to the inventory service and then deletes the
cached entry, so the next get_product() in any process reloads the product with
its new price and re-caches it.

When several processes miss the same key at the same time, the first one claims
the population (SET NX marker) and fetches upstream once; the others wait for
the key to appear, so a cold start costs one upstream read per product.
"""
import json
import time
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"

VALKEY_HOST = "127.0.0.1"
VALKEY_PORT = 6379
CACHE_TTL_SECONDS = 300  # cached products expire after 5 minutes
POPULATION_LOCK_TTL = 3  # safety bound for the "someone else is fetching" marker
_WAIT_FOR_POPULATION = 2.0  # seconds to wait for another process's fetch before fetching ourselves
_POPULATING = "-"  # placeholder value while a product is being fetched

_client_instance = None


def _key(product_id: int) -> str:
    return f"catalog:product:{int(product_id)}"


def _cache() -> "valkey.client.Valkey":
    global _client_instance
    if _client_instance is None:
        # decode_responses=True so cached products come back as str, ready for json.loads
        _client_instance = valkey.Valkey(host=VALKEY_HOST, port=VALKEY_PORT,
                                         decode_responses=True,
                                         socket_timeout=2, socket_connect_timeout=2)
    return _client_instance


def _fetch_product(product_id: int) -> dict:
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        product = json.load(resp)
    try:
        # Authoritative read: store it, refreshing the 5-minute expiry.
        _cache().set(_key(product_id), json.dumps(product), ex=CACHE_TTL_SECONDS)
    except Exception:
        pass  # a cache outage must not make reads fail
    return product


def get_product(product_id: int) -> dict:
    key = _key(product_id)
    try:
        cached = _cache().get(key)
        if cached is not None and cached != _POPULATING:
            return json.loads(cached)
        # Miss: claim the right to fetch so concurrent misses cause one upstream read.
        claimed = _cache().set(key, _POPULATING, ex=POPULATION_LOCK_TTL, nx=True)
    except Exception:
        claimed = True  # no usable cache: just read from the inventory service
    if claimed:
        return _fetch_product(product_id)
    # Another process is fetching this product; wait for it to appear in the cache.
    deadline = time.monotonic() + _WAIT_FOR_POPULATION
    while time.monotonic() < deadline:
        time.sleep(0.02)
        try:
            cached = _cache().get(key)
        except Exception:
            cached = None
        if cached is not None and cached != _POPULATING:
            return json.loads(cached)
    return _fetch_product(product_id)  # populator died or cache vanished: fetch ourselves


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()
    # The cached copy is now stale: drop it so the next get_product() in any
    # process reloads the new price and re-caches it (bounded anyway by the
    # 5-minute TTL). If Valkey is unreachable, the entry expires on its own.
    try:
        _cache().delete(_key(product_id))
    except Exception:
        pass
