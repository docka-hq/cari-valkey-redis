#!/usr/bin/env python3
"""Semantic search over Valkey (Valkey Search module, kNN / cosine).

stdin:  a JSON list of 64 floats (the query embedding)
stdout: ids of the 5 most similar articles, most similar first

The similarity search runs entirely inside Valkey (FT.SEARCH with a KNN
vector query); this script only ships the query vector to the server and
prints the ids of the documents Valkey returns.
"""
import json
import struct
import sys

import redis

INDEX = "idx:articles"
TOP_K = 5


def main() -> None:
    query = json.load(sys.stdin)
    if len(query) != 64:
        sys.exit(f"expected 64 floats, got {len(query)}")
    # Raw little-endian FLOAT32 blob, the format the index stores.
    vec = struct.pack("<64f", *query)

    r = redis.Redis(host="127.0.0.1", port=6379, decode_responses=False)
    # Pure kNN query: Valkey ranks by cosine distance and returns the
    # top TOP_K documents, nearest first.
    res = r.execute_command(
        "FT.SEARCH", INDEX,
        f"=>[KNN {TOP_K} @embedding $vec]",
        "PARAMS", "2", "vec", vec,
        "DIALECT", "2",
        "LIMIT", "0", TOP_K,
    )

    for key in res[1::2]:  # [total, key1, doc1, key2, doc2, ...]
        print(key.decode().split(":", 1)[1])


if __name__ == "__main__":
    main()
