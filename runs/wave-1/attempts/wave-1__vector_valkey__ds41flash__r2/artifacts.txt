#!/usr/bin/env python3
"""Semantic search over the knowledge-base articles stored in Valkey.

Usage:  python3 /app/search.py < query.json

query.json contains a single query embedding: a JSON list of 64 floats.
The script prints the ids of the 5 most similar articles, one per line,
most similar first.

The similarity search itself runs entirely inside Valkey (the valkey-search
module's FT.SEARCH KNN query over a cosine-distance vector index); this script
only encodes the query vector and forwards it.
"""
import json
import struct
import sys

import redis

INDEX = "articles_idx"
VECTOR_FIELD = "embedding"
DIM = 64
K = 5


def main() -> None:
    query = json.load(sys.stdin)
    if len(query) != DIM:
        raise SystemExit(f"expected a {DIM}-dimension query vector, got {len(query)}")

    # FLOAT32 little-endian blob, matching the index's declared type.
    blob = struct.pack(f"<{DIM}f", *(float(x) for x in query))

    client = redis.Redis(host="127.0.0.1", port=6379)

    # Vector results are returned in ascending distance order by default.
    result = client.execute_command(
        "FT.SEARCH", INDEX,
        f"*=>[KNN {K} @{VECTOR_FIELD} $vec AS score]",
        "PARAMS", "2", "vec", blob,
        "RETURN", "1", "id",
        "DIALECT", "2",
        "LIMIT", "0", str(K),
    )

    # Response shape: [total, key1, [field, value, ...], key2, [...], ...]
    for i in range(1, len(result), 2):
        fields = result[i + 1]
        fields = dict(zip(fields[::2], fields[1::2]))
        print(fields[b"id"].decode())


if __name__ == "__main__":
    main()
