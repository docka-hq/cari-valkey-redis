"""Print the five closest article IDs for a 64-dimensional query vector."""
import json
import struct
import sys

import redis


def main():
    vector = json.load(sys.stdin)
    if not isinstance(vector, list) or len(vector) != 64:
        raise ValueError('Expected a JSON list of 64 numbers')
    blob = struct.pack('<64d', *vector)

    client = redis.Redis(host='127.0.0.1', port=6379)
    result = client.execute_command(
        'FT.SEARCH', 'kb_articles', '*=>[KNN 5 @embedding $query AS distance]',
        'PARAMS', 2, 'query', blob,
        'SORTBY', 'distance', 'ASC',
        'RETURN', 1, 'id',
        'LIMIT', 0, 5,
        'DIALECT', 2,
    )
    if isinstance(result, dict):
        for hit in result[b'results']:
            print(hit[b'extra_attributes'][b'id'].decode('utf-8'))
    else:
        for fields in result[2::2]:
            values = dict(zip(fields[::2], fields[1::2]))
            print(values[b'id'].decode('utf-8'))


if __name__ == '__main__':
    main()
