#!/usr/bin/env python3
"""Return the five nearest knowledge-base article IDs using Valkey vector search."""
import json
import math
import struct
import sys

import valkey


def main():
    vector = json.load(sys.stdin)
    if not isinstance(vector, list) or len(vector) != 64 or any(
        not isinstance(x, (float, int)) or isinstance(x, bool) or not math.isfinite(x)
        for x in vector
    ):
        raise ValueError('query must be a JSON list of 64 finite numbers')
    blob = struct.pack('<64f', *vector)
    client = valkey.Valkey(host='127.0.0.1', port=6379, decode_responses=False)
    result = client.execute_command(
        'FT.SEARCH', 'articles_idx', '*=>[KNN 5 @embedding $vec AS distance]',
        'PARAMS', 2, 'vec', blob,
        'LIMIT', 0, 5, 'NOCONTENT', 'DIALECT', 2,
    )
    for key in result[1:]:
        print(key.decode('utf-8').removeprefix('article:'))


if __name__ == '__main__':
    main()
