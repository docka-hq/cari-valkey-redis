"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in Redis (127.0.0.1:6379), shared by all storefront
processes, for CACHE_TTL seconds.  update_price() invalidates the cached
entry so the next get_product() in any process sees the new price.

To avoid a race where a reader fetches the old product from the inventory
service, a writer updates the price and invalidates, and then the reader
writes the stale product back into the cache, every product has a version
counter in Redis.  update_price() bumps the version; get_product() only
stores its result if the version is unchanged since before it fetched.

If Redis is unavailable, get_product() falls back to the inventory service.
"""
import json
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"
REDIS_URL = "redis://127.0.0.1:6379/0"
CACHE_TTL = 300  # seconds (5 minutes)
_KEY_PREFIX = "catalog:product:"

_redis = redis.Redis.from_url(
    REDIS_URL, socket_timeout=0.5, socket_connect_timeout=0.5
)

# KEYS[1] = data key, KEYS[2] = version key
# ARGV[1] = version observed before fetch, ARGV[2] = value, ARGV[3] = ttl
_SET_IF_VERSION = _redis.register_script(
    """
    local v = redis.call('GET', KEYS[2]) or '0'
    if v == ARGV[1] then
        redis.call('SET', KEYS[1], ARGV[2], 'EX', tonumber(ARGV[3]))
        return 1
    end
    return 0
    """
)


def _keys(product_id: int):
    base = f"{_KEY_PREFIX}{int(product_id)}"
    return base, base + ":ver"


def _fetch(product_id: int) -> dict:
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    key, ver_key = _keys(product_id)
    try:
        cached, version = _redis.mget(key, ver_key)
    except redis.RedisError:
        return _fetch(product_id)
    if cached is not None:
        return json.loads(cached)

    product = _fetch(product_id)
    try:
        _SET_IF_VERSION(
            keys=[key, ver_key],
            args=[(version or b"0").decode(), json.dumps(product), CACHE_TTL],
        )
    except redis.RedisError:
        pass
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate after the inventory service has accepted the change.
    # Errors are propagated: silently skipping invalidation would leave a
    # stale price cached for up to CACHE_TTL.
    key, ver_key = _keys(product_id)
    pipe = _redis.pipeline(transaction=True)
    pipe.incr(ver_key)
    pipe.delete(key)
    pipe.execute()
