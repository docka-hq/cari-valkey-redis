$ cat /app/search.py
import json, struct, sys, redis
q = json.load(sys.stdin)
if len(q) != 64: sys.exit("query must have 64 floats")
r = redis.Redis(host='127.0.0.1', port=6379, decode_responses=False, protocol=2)
res = r.execute_command('FT.SEARCH', 'idx:articles', '*=>[KNN 5 @embedding $v AS score]',
    'PARAMS', '2', 'v', struct.pack('<64f', *q),
    'SORTBY', 'score', 'ASC', 'RETURN', '2', 'id', 'score', 'LIMIT', '0', '5', 'DIALECT', '2')
for i in range(1, len(res), 2):
    f = dict(zip(res[i+1][::2], res[i+1][1::2]))
    print(f[b'id'].decode())
