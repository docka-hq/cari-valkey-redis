"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Product reads are served from a shared Valkey cache so that every storefront
process can reuse them instead of hitting the (slow) inventory service.

Cache entries live for 5 minutes.  A price update invalidates the entry for
that product, so the next read in any process goes back to the inventory
service and sees the new price.
"""
import json
import os
import time
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"

VALKEY_HOST = "127.0.0.1"
VALKEY_PORT = 6379
CACHE_TTL = 300  # seconds (5 minutes)

# How long to wait for the per-product lock and how long the lock may live.
LOCK_TIMEOUT = 5.0       # seconds a caller waits for the lock
LOCK_TTL_MS = 10_000     # safety TTL so a crashed holder cannot wedge the key

# Release the lock only if we still own it (atomic compare-and-delete).
_UNLOCK_SCRIPT = """
if redis.call('get', KEYS[1]) == ARGV[1] then
    return redis.call('del', KEYS[1])
end
return 0
"""

_client = None
_client_pid = None


def _cache():
    """Return a process-local Valkey client (re-created after a fork)."""
    global _client, _client_pid
    pid = os.getpid()
    if _client is None or _client_pid != pid:
        _client = valkey.Valkey(
            host=VALKEY_HOST,
            port=VALKEY_PORT,
            decode_responses=True,
            socket_timeout=5,
            socket_connect_timeout=5,
        )
        _client_pid = pid
    return _client


def _cache_key(product_id: int) -> str:
    return f"catalog:product:{int(product_id)}"


def _lock_key(product_id: int) -> str:
    return f"catalog:product:{int(product_id)}:lock"


def _acquire_lock(client, key):
    """Best-effort per-product lock. Returns a token, or None on timeout."""
    token = f"{os.getpid()}-{time.monotonic()}"
    deadline = time.monotonic() + LOCK_TIMEOUT
    while True:
        if client.set(key, token, nx=True, px=LOCK_TTL_MS):
            return token
        if time.monotonic() >= deadline:
            return None
        time.sleep(0.02)


def _release_lock(client, key, token):
    if token is None:
        return
    try:
        client.eval(_UNLOCK_SCRIPT, 1, key, token)
    except Exception:
        pass


def _fetch_from_inventory(product_id: int) -> dict:
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    key = _cache_key(product_id)

    try:
        client = _cache()
        cached = client.get(key)
        if cached is not None:
            return json.loads(cached)
    except Exception:
        # Cache unavailable: fall back to the inventory service.
        return _fetch_from_inventory(product_id)

    # Cache miss. Serialize the refresh per product so a burst of callers only
    # results in a single inventory request (avoids a stampede).
    lock = _lock_key(product_id)
    token = None
    try:
        token = _acquire_lock(client, lock)
    except Exception:
        token = None

    try:
        # Another process may have populated the cache while we waited.
        cached = client.get(key)
        if cached is not None:
            return json.loads(cached)

        product = _fetch_from_inventory(product_id)
        try:
            client.set(key, json.dumps(product), ex=CACHE_TTL)
        except Exception:
            pass
        return product
    finally:
        _release_lock(client, lock, token)


def update_price(product_id: int, price: float) -> None:
    key = _cache_key(product_id)

    try:
        client = _cache()
        lock = _lock_key(product_id)
        token = _acquire_lock(client, lock)
    except Exception:
        client = None
        lock = None
        token = None

    try:
        body = json.dumps({"price": price}).encode()
        req = urllib.request.Request(
            f"{INVENTORY_URL}/products/{int(product_id)}",
            data=body,
            method="PUT",
            headers={"Content-Type": "application/json"},
        )
        with urllib.request.urlopen(req, timeout=10) as resp:
            resp.read()

        # Invalidate the shared entry so every process re-reads the new price.
        if client is not None:
            try:
                client.delete(key)
            except Exception:
                pass
    finally:
        if client is not None:
            _release_lock(client, lock, token)
