#!/usr/bin/env python3
"""Semantic search over knowledge-base articles, served entirely by Redis.

Usage: python3 /app/search.py < query.json
where query.json is a JSON list of 64 floats (the query embedding).

All article embeddings live in Redis hashes (article:<id>, field 'emb'),
indexed by idx:articles: a 64-dim FLOAT32 vector index over the 1,000
articles. The nearest-neighbor query, the distance metric and the
ranking of the top 5 all run inside Redis (FT.AGGREGATE with a KNN
vector step, sorted by ascending distance). This script only packs the
query vector and prints the ids Redis returns.
"""
import json
import struct
import sys

import redis

INDEX = "idx:articles"
PREFIX = "article:"
DIM = 64
K = 5


def main() -> None:
    query = json.load(sys.stdin)
    if not isinstance(query, list) or len(query) != DIM:
        sys.exit(f"error: expected a JSON list of {DIM} floats")

    blob = struct.pack(f"<{DIM}f", *query)  # little-endian float32

    r = redis.Redis(host="127.0.0.1", port=6379)
    res = r.execute_command(
        "FT.AGGREGATE", INDEX,
        f"*=>[KNN {K} @emb $vec AS score]",
        "PARAMS", 2, "vec", blob,
        "DIALECT", 2,
        "LOAD", 2, "@id", "@score",
        "SORTBY", 2, "@score", "ASC",
    )

    rows = res[b"results"] if isinstance(res, dict) else res[1:]
    for row in rows:
        attrs = row[b"extra_attributes"] if isinstance(res, dict) else row[1]
        key = attrs[b"id"] if isinstance(attrs, dict) else attrs[attrs.index(b"id") + 1]
        print(key.decode())


if __name__ == "__main__":
    main()
