#!/usr/bin/env python3
"""Semantic search over the knowledge-base articles stored in Valkey.

Usage:
    python3 /app/search.py < query.json

query.json holds a single query embedding: a JSON list of 64 floats
(a JSON object with an "embedding" key is also accepted).

Prints the ids of the 5 most similar articles (cosine similarity), most
similar first, one per line.

How it works
------------
All 1,000 articles live in Valkey as JSON documents under keys
`kb:vec:<id>` (field `$.embedding`), indexed by the `kb:articles` vector
index created with:

    FT.CREATE kb:articles ON JSON PREFIX 1 kb:vec: SCHEMA $.embedding
        AS embedding VECTOR FLAT 6 TYPE FLOAT32 DIM 64 DISTANCE_METRIC COSINE

This script sends the query vector to Valkey and runs a KNN query *inside*
the server:

    FT.SEARCH kb:articles '*=>[KNN 5 @embedding $vec AS distance]'
        PARAMS 2 vec <blob> DIALECT 2

The FLAT algorithm performs an exact brute-force scan, so the top-5 by
cosine distance (= 1 - cosine similarity) is the true top-5. Article
titles/metadata are kept separately in hashes `kb:meta:<id>`; this script
only needs the ids. Nothing is read from /data/articles.jsonl and no
similarities are computed locally.
"""

import json
import sys

import numpy as np
from valkey import Valkey

INDEX_NAME = "kb:articles"
KEY_PREFIX = "kb:vec:"
DIM = 64
K = 5
QUERY_TEMPLATE = f"*=>[KNN {K} @embedding $vec AS distance]"


def read_query_vector(stream) -> np.ndarray:
    """Read one 64-float embedding from JSON text and return it as float32."""
    obj = json.load(stream)
    if isinstance(obj, dict):  # tolerate {"embedding": [...]}
        obj = obj["embedding"]
    if not isinstance(obj, list) or len(obj) != DIM:
        raise ValueError(f"query must be a JSON list of {DIM} floats")
    return np.asarray(obj, dtype=np.float32)


def search(valkey: Valkey, query_vec: np.ndarray) -> list[tuple[str, float]]:
    """Run the KNN query inside Valkey; return [(article_id, distance), ...]."""
    raw = valkey.execute_command(
        "FT.SEARCH", INDEX_NAME, QUERY_TEMPLATE,
        "PARAMS", 2, "vec", query_vec.tobytes(),
        "DIALECT", 2,
    )
    # Reply layout: [total, key1, attrs1, key2, attrs2, ...] where attrs is a
    # flat list of (name, value) pairs; the score pair uses our alias
    # ("distance") and the JSON document itself arrives under the "$" name.
    total = int(raw[0])
    results: list[tuple[str, float]] = []
    for i in range(total):
        key = raw[1 + 2 * i]
        attrs = raw[2 + 2 * i] or []
        article_id = key.decode()
        if article_id.startswith(KEY_PREFIX):
            article_id = article_id[len(KEY_PREFIX):]
        distance = None
        for j in range(0, len(attrs) - 1, 2):
            name = attrs[j]
            if isinstance(name, bytes) and name.decode() == "$":
                continue
            try:
                distance = float(attrs[j + 1])
            except (TypeError, ValueError):
                distance = None
            break
        if distance is None:
            distance = float(i)  # fall back to reply order
        results.append((article_id, distance))
    return results


def main() -> int:
    try:
        query_vec = read_query_vector(sys.stdin)
    except Exception as exc:
        print(f"error reading query: {exc}", file=sys.stderr)
        return 1
    try:
        valkey = Valkey(host="127.0.0.1", port=6379, db=0, decode_responses=False)
        results = search(valkey, query_vec)
    except Exception as exc:
        print(f"error querying Valkey: {exc}", file=sys.stderr)
        return 1
    results.sort(key=lambda item: item[1])  # cosine distance: lower is better
    for article_id, _ in results[:K]:
        print(article_id)
    return 0


if __name__ == "__main__":
    sys.exit(main())
