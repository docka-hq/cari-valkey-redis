"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in Valkey (shared by all storefront processes) for 5 minutes.
update_price() invalidates the cached entry. A per-product version counter
prevents a reader that fetched the old data concurrently with an update from
writing a stale entry back into the cache after the invalidation.
"""
import json
import logging
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"
VALKEY_HOST = "127.0.0.1"
VALKEY_PORT = 6379
CACHE_TTL_SECONDS = 300
VERSION_TTL_SECONDS = 86400

log = logging.getLogger(__name__)

_client = valkey.Valkey(
    host=VALKEY_HOST,
    port=VALKEY_PORT,
    socket_timeout=0.5,
    socket_connect_timeout=0.5,
)

# Store the value only if the product's version hasn't changed since the
# reader started fetching from the inventory service.
_SET_IF_VERSION = _client.register_script(
    """
    local cur = redis.call('GET', KEYS[2])
    if (cur or '') == ARGV[1] then
        redis.call('SET', KEYS[1], ARGV[2], 'EX', ARGV[3])
        return 1
    end
    return 0
    """
)


def _cache_key(product_id: int) -> str:
    return f"catalog:product:{int(product_id)}"


def _version_key(product_id: int) -> str:
    return f"catalog:product:{int(product_id)}:ver"


def _fetch(product_id: int) -> dict:
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    key, vkey = _cache_key(product_id), _version_key(product_id)
    version = None
    try:
        cached, version = _client.mget(key, vkey)
        if cached is not None:
            return json.loads(cached)
    except valkey.ValkeyError as exc:
        log.warning("Valkey unavailable, reading inventory directly: %s", exc)
        return _fetch(product_id)

    product = _fetch(product_id)
    try:
        _SET_IF_VERSION(
            keys=[key, vkey],
            args=[(version or b"").decode(), json.dumps(product), CACHE_TTL_SECONDS],
        )
    except valkey.ValkeyError as exc:
        log.warning("Could not cache product %s: %s", product_id, exc)
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate after the write succeeded; bump version so in-flight readers
    # holding pre-update data can't repopulate the cache with it.
    pipe = _client.pipeline(transaction=True)
    pipe.incr(_version_key(product_id))
    pipe.expire(_version_key(product_id), VERSION_TTL_SECONDS)
    pipe.delete(_cache_key(product_id))
    pipe.execute()
