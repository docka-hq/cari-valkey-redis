"""Return the five closest knowledge-base articles using Valkey's cosine vector index."""
import json
import struct
import sys

import redis


def main():
    embedding = json.load(sys.stdin)
    if not isinstance(embedding, list) or len(embedding) != 64:
        raise ValueError('Expected a JSON list of 64 floats')
    vector = struct.pack('<64f', *embedding)
    client = redis.Redis(host='127.0.0.1', port=6379)
    result = client.execute_command(
        'FT.SEARCH', 'kb:idx', '*=>[KNN 5 @embedding $query]',
        'PARAMS', 2, 'query', vector,
        'RETURN', 1, 'id', 'LIMIT', 0, 5, 'DIALECT', 2,
    )
    if result[0] != 5:
        raise RuntimeError(f'Expected 5 search results, got {result[0]}')
    for fields in result[2::2]:
        values = dict(zip(fields[::2], fields[1::2]))
        print(values[b'id'].decode('utf-8'))


if __name__ == '__main__':
    main()
