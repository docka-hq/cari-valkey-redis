#!/usr/bin/env python3
"""Semantic search over knowledge-base articles using Valkey Search (FT.SEARCH).

Usage: python3 /app/search.py < query.json
  query.json contains one query embedding (a JSON list of 64 floats).
Prints the ids of the 5 most similar articles (cosine similarity), one per
line, most similar first. The similarity ranking is computed entirely inside
Valkey by the KNN vector query; this script only forwards the query vector.
"""
import json
import struct
import sys

from valkey import Valkey

INDEX = "idx:articles"
KEY_PREFIX = "article:"
K = 5
DIM = 64


def main() -> None:
    query_embedding = json.load(sys.stdin)
    if len(query_embedding) != DIM:
        sys.exit(f"query embedding must have {DIM} dimensions")
    query_blob = struct.pack(f"<{DIM}f", *query_embedding)

    r = Valkey(host="127.0.0.1", port=6379)
    # KNN query: Valkey ranks all indexed articles by cosine distance and
    # returns the K closest ones, already ordered nearest-first.
    reply = r.execute_command(
        "FT.SEARCH", INDEX,
        f"*=>[KNN {K} @embedding $query_vector AS vec_distance]",
        "PARAMS", 2, "query_vector", query_blob,
        "NOCONTENT",
        "DIALECT", 2,
    )
    # reply = [total_hits, key1, key2, ...] because of NOCONTENT
    for item in reply[1 : 1 + K]:
        key = item.decode() if isinstance(item, bytes) else item
        print(key[len(KEY_PREFIX):] if key.startswith(KEY_PREFIX) else key)


if __name__ == "__main__":
    main()
