#!/usr/bin/env python3
"""Find the five closest knowledge-base articles using Redis vector search."""
import json
import struct
import sys

import redis


def main():
    vector = json.load(sys.stdin)
    if not isinstance(vector, list) or len(vector) != 64:
        raise ValueError('expected a JSON list of 64 numbers')
    query = struct.pack('<64d', *vector)
    connection = redis.Redis(host='127.0.0.1', port=6379)
    response = connection.execute_command(
        'FT.SEARCH', 'kb_articles',
        '*=>[KNN 5 @embedding $query_vector AS distance]',
        'PARAMS', '2', 'query_vector', query,
        'SORTBY', 'distance', 'ASC',
        'RETURN', '1', 'id',
        'DIALECT', '2',
    )
    for result in response[b'results']:
        print(result[b'extra_attributes'][b'id'].decode('utf-8'))


if __name__ == '__main__':
    main()
