"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service
"""
import json
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"

# All storefront processes use the same cache and per-product locks.
_cache = valkey.Valkey(host="127.0.0.1", port=6379, decode_responses=True)
_CACHE_TTL_SECONDS = 300


def _keys(product_id: int) -> tuple[str, str]:
    product_id = int(product_id)
    return f"catalog:product:{product_id}", f"catalog:product:{product_id}:lock"


def get_product(product_id: int) -> dict:
    product_id = int(product_id)
    key, lock_key = _keys(product_id)
    cached = _cache.get(key)
    if cached is not None:
        return json.loads(cached)

    # Serialize fills with price updates so a slow read cannot repopulate
    # the cache with an old price after an update has invalidated it.
    with _cache.lock(lock_key, timeout=30):
        cached = _cache.get(key)
        if cached is not None:
            return json.loads(cached)
        with urllib.request.urlopen(f"{INVENTORY_URL}/products/{product_id}", timeout=10) as resp:
            product = json.load(resp)
        _cache.set(key, json.dumps(product), ex=_CACHE_TTL_SECONDS)
        return product


def update_price(product_id: int, price: float) -> None:
    product_id = int(product_id)
    key, lock_key = _keys(product_id)
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{product_id}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with _cache.lock(lock_key, timeout=30):
        with urllib.request.urlopen(req, timeout=10) as resp:
            resp.read()
        # Invalidate only after the service has confirmed the write. The next
        # read (in this process or another) will fetch the updated product.
        _cache.delete(key)
