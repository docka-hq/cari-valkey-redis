#!/usr/bin/env python3
"""Semantic search over the article vector index in Redis.

Usage:  python3 /app/search.py < query.json

query.json contains one query embedding (a JSON list of 64 floats).
Prints the ids of the 5 most similar articles (cosine similarity),
one per line, most similar first.

The similarity search runs inside Redis (RediSearch KNN query);
this script neither reads the article file nor computes similarities.
"""
import json
import struct
import sys

import redis

INDEX = "articles"
TOP_K = 5

def main() -> None:
    query = json.load(sys.stdin)  # list of 64 floats
    vec = struct.pack("<%df" % len(query), *query)

    r = redis.Redis(host="127.0.0.1", port=6379, decode_responses=True)
    res = r.execute_command(
        "FT.SEARCH", INDEX,
        "*=>[KNN %d @embedding $vec AS score]" % TOP_K,
        "PARAMS", "2", "vec", vec,
        "SORTBY", "score", "ASC",
        "RETURN", "1", "id",
        "DIALECT", "2",
    )

    results = res["results"] if isinstance(res, dict) else res[2::2]
    for item in results:
        if isinstance(item, dict):
            sys.stdout.write(item["extra_attributes"]["id"] + "\n")

if __name__ == "__main__":
    main()
