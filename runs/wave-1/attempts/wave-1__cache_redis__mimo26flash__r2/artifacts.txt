"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in the shared Redis server (127.0.0.1:6379) so that every
storefront process reuses recently read products instead of calling the
inventory service.  Cache entries expire after 5 minutes, and update_price()
invalidates the entry so the next read (in any process) sees the new price.
"""
import json
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"

REDIS_HOST = "127.0.0.1"
REDIS_PORT = 6379
CACHE_TTL_SECONDS = 300  # 5 minutes

# One shared connection pool per process; Redis is the cross-process cache.
_redis_client = redis.Redis(
    host=REDIS_HOST,
    port=REDIS_PORT,
    decode_responses=True,
    socket_timeout=2,
    socket_connect_timeout=2,
)


def _cache_key(product_id: int) -> str:
    return f"product:{int(product_id)}"


def _fetch_from_inventory(product_id: int) -> dict:
    with urllib.request.urlopen(
        f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10
    ) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    key = _cache_key(product_id)
    try:
        cached = _redis_client.get(key)
    except redis.RedisError:
        cached = None
    if cached is not None:
        return json.loads(cached)

    product = _fetch_from_inventory(product_id)

    try:
        # NX + EX: only the first caller populates the entry, with a 5 minute TTL.
        _redis_client.set(key, json.dumps(product), ex=CACHE_TTL_SECONDS, nx=True)
    except redis.RedisError:
        pass  # caching is best-effort; still return the fresh product
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    key = _cache_key(product_id)
    # Invalidate the shared cache so the next get_product() -- in any
    # process -- refetches and sees the new price.  A concurrent stale
    # reader may repopulate the key with an old value, so after deleting we
    # also write back a freshly fetched copy (plain SET overwrites anything
    # a stale writer put there; get_product() uses NX and won't clobber it).
    try:
        _redis_client.delete(key)
    except redis.RedisError:
        return
    try:
        product = _fetch_from_inventory(product_id)
        _redis_client.set(key, json.dumps(product), ex=CACHE_TTL_SECONDS)
    except Exception:
        # Refresh is best-effort; the key is already deleted, so the next
        # get_product() call will fetch the new price from inventory.
        pass
