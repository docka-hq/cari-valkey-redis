#!/usr/bin/env python3
"""Usage: python3 /app/search.py < query.json  -> top-5 article ids by cosine similarity (KNN in Valkey Search)."""
import json, sys
import numpy as np
import valkey

K = 5
query = json.load(sys.stdin)
vec = np.asarray(query, dtype=np.float32)
if vec.shape != (64,):
    sys.exit(f"expected 64-dim embedding, got shape {vec.shape}")

r = valkey.Valkey(host="127.0.0.1", port=6379)
res = r.execute_command(
    "FT.SEARCH", "articles_idx", f"*=>[KNN {K} @embedding $vec AS score]",
    "PARAMS", "2", "vec", vec.tobytes(),
    "RETURN", "2", "id", "score",
    "LIMIT", "0", str(K),
    "DIALECT", "2",
)
hits = []
for i in range(1, len(res), 2):
    key = res[i].decode()
    f = res[i + 1]
    d = {f[j].decode(): f[j + 1].decode() for j in range(0, len(f), 2)}
    hits.append((float(d["score"]), d.get("id") or key.split(":", 1)[1]))
hits.sort(key=lambda h: h[0])  # cosine distance ascending = most similar first
for _, aid in hits[:K]:
    print(aid)
