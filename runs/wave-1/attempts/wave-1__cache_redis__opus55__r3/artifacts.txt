"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in Redis (shared by all storefront processes) for
CACHE_TTL seconds. update_price() invalidates the cached entry so the next
get_product() in any process sees the new price. A per-product version
counter prevents a reader that fetched stale data *before* an update from
re-populating the cache with that stale data *after* the invalidation.
If Redis is unavailable, calls fall through to the inventory service.
"""
import json
import logging
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"
REDIS_URL = "redis://127.0.0.1:6379/0"
CACHE_TTL = 300  # seconds (5 minutes)

log = logging.getLogger(__name__)

_redis = redis.Redis.from_url(
    REDIS_URL, socket_timeout=0.5, socket_connect_timeout=0.5, decode_responses=True
)

# Store the product only if no update_price() happened since we read the version.
_SET_IF_VERSION = _redis.register_script(
    """
    local cur = redis.call('GET', KEYS[2]) or '0'
    if cur == ARGV[1] then
        redis.call('SET', KEYS[1], ARGV[2], 'EX', ARGV[3])
        return 1
    end
    return 0
    """
)


def _key(product_id: int) -> str:
    return f"catalog:product:{int(product_id)}"


def _ver_key(product_id: int) -> str:
    return f"catalog:product:{int(product_id)}:ver"


def _fetch(product_id: int) -> dict:
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    key, vkey = _key(product_id), _ver_key(product_id)
    version = None
    try:
        cached, version = _redis.mget(key, vkey)
        if cached is not None:
            return json.loads(cached)
        version = version or "0"
    except redis.RedisError as exc:
        log.warning("redis read failed, bypassing cache: %s", exc)
        version = None

    product = _fetch(product_id)

    if version is not None:
        try:
            _SET_IF_VERSION(keys=[key, vkey], args=[version, json.dumps(product), CACHE_TTL])
        except redis.RedisError as exc:
            log.warning("redis write failed: %s", exc)
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate: bump the version (blocks in-flight stale writes) and drop the entry.
    key, vkey = _key(product_id), _ver_key(product_id)
    pipe = _redis.pipeline(transaction=True)
    pipe.incr(vkey)
    pipe.delete(key)
    pipe.execute()
