import sys, json, struct, redis
q = json.load(sys.stdin)
r = redis.Redis(host='127.0.0.1', port=6379, protocol=2)
res = r.execute_command('FT.SEARCH', 'idx:articles', '*=>[KNN 5 @embedding $v AS score]',
    'PARAMS', '2', 'v', struct.pack(f'<{len(q)}f', *q),
    'SORTBY', 'score', 'ASC', 'RETURN', '2', 'id', 'score', 'LIMIT', '0', '5', 'DIALECT', '2')
if isinstance(res, dict):  # RESP3 reply shape
    rows = [x[b'extra_attributes'] for x in res[b'results']]
else:                      # RESP2 reply shape
    rows = [dict(zip(res[i][::2], res[i][1::2])) for i in range(2, len(res), 2)]
for f in rows:
    print(f[b'id'].decode())
