#!/usr/bin/env python3
"""Semantic search over the Valkey-hosted article embeddings.

Usage:  python3 /app/search.py < query.json

Reads one query embedding (JSON list of 64 floats) from stdin and prints the
ids of the 5 most similar articles (cosine similarity), most similar first.
The similarity search itself is executed inside Valkey via an exact KNN query
(FT.SEARCH, FLAT index, COSINE distance) -- no local computation of
similarities and no access to the source data file.
"""
import json
import struct
import sys

import redis

INDEX = "articles_idx"
TOP_K = 5


def main() -> None:
    embedding = json.load(sys.stdin)
    if len(embedding) != 64:
        raise SystemExit(f"expected 64 floats, got {len(embedding)}")
    vec = struct.pack("<64f", *embedding)

    r = redis.Redis(host="127.0.0.1", port=6379, decode_responses=False)

    # Exact k-nearest-neighbour search run server-side inside Valkey.
    # COSINE distance = 1 - cosine similarity, returned ascending,
    # which is exactly cosine similarity descending.
    query = f"*=>[KNN {TOP_K} @embedding $vec]"
    res = r.execute_command(
        "FT.SEARCH", INDEX, query,
        "PARAMS", "2", "vec", vec,
        "RETURN", "1", "id",
        "DIALECT", "2",
        "LIMIT", "0", str(TOP_K),
        "NOCONTENT",
    )

    # reply: [total_count, key1, key2, ...] with NOCONTENT
    keys = res[1:]
    for key in keys:
        # keys look like "article:kb-0123"; prefer the indexed `id` field
        art_id = r.hget(key, "id")
        print(art_id.decode() if art_id else key.decode().split(":", 1)[-1])


if __name__ == "__main__":
    main()
