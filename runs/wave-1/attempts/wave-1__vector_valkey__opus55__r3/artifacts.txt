#!/usr/bin/env python3
import sys, json, struct, valkey

def main():
    q = json.load(sys.stdin)
    if len(q) != 64:
        sys.exit(f"expected 64-dim embedding, got {len(q)}")
    r = valkey.Valkey(host='127.0.0.1', port=6379)
    res = r.execute_command(
        'FT.SEARCH', 'articles_idx', '*=>[KNN 5 @embedding $vec AS score]',
        'PARAMS', 2, 'vec', struct.pack('<64f', *map(float, q)),
        'RETURN', 2, 'id', 'score', 'LIMIT', 0, 5, 'DIALECT', 2)
    hits = []
    for i in range(1, len(res), 2):
        f = res[i + 1]
        d = {f[j].decode(): f[j + 1].decode() for j in range(0, len(f), 2)}
        hits.append((float(d['score']), d['id']))
    hits.sort(key=lambda h: h[0])
    for _, aid in hits[:5]:
        print(aid)

if __name__ == '__main__':
    main()
