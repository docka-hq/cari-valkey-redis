#!/usr/bin/env python3
"""Semantic search over the knowledge-base articles stored in Valkey.

Reads a query embedding (JSON list of 64 floats) from stdin, runs a
cosine-similarity vector search inside Valkey (valkey-search module,
FT.SEARCH over a FLAT vector index), and prints the ids of the 5 most
similar articles, most similar first.
"""
import json
import struct
import sys

try:
    import valkey as client_lib
except ImportError:
    import redis as client_lib

TOP_K = 5


def main():
    query = json.load(sys.stdin)
    # binary float32 encoding of the query vector, as expected by FT.SEARCH
    vec = struct.pack("<%df" % len(query), *query)

    r = client_lib.Valkey(host="127.0.0.1", port=6379, decode_responses=False)
    res = r.execute_command(
        "FT.SEARCH", "articles", "*=>[KNN 5 @embedding $vec]",
        "PARAMS", "2", "vec", vec,
        "LIMIT", "0", str(TOP_K),
        "RETURN", "1", "id",
        "DIALECT", "2",
    )

    # reply: [total, key1, [field, value, ...], key2, ...]
    out = sys.stdout
    it = iter(res[1:])
    for key in it:
        fields = next(it)
        attrs = dict(zip(fields[0::2], fields[1::2]))
        out.write(attrs[b"id"].decode() + "\n")


if __name__ == "__main__":
    main()
