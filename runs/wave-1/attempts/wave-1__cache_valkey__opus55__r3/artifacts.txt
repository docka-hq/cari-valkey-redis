"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in Valkey (shared by all storefront processes) for
CACHE_TTL seconds. update_price() invalidates the cached entry, and a
per-product version counter prevents a concurrent reader from writing a
stale (pre-update) product back into the cache.
"""
import json
import logging
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"
VALKEY_URL = "valkey://127.0.0.1:6379/0"
CACHE_TTL = 300  # seconds

log = logging.getLogger(__name__)

_client = valkey.Valkey.from_url(
    VALKEY_URL, socket_timeout=0.5, socket_connect_timeout=0.5, decode_responses=True
)

# Store the value only if the product's version hasn't changed since the
# reader looked at it (i.e. no update_price() happened during the fetch).
_SET_IF_VERSION = _client.register_script(
    """
    local cur = redis.call('GET', KEYS[2]) or '0'
    if cur == ARGV[1] then
        redis.call('SET', KEYS[1], ARGV[2], 'EX', ARGV[3])
        return 1
    end
    return 0
    """
)


def _keys(product_id: int):
    tag = f"{{catalog:product:{int(product_id)}}}"  # same hash slot in cluster mode
    return f"{tag}:data", f"{tag}:ver"


def _fetch(product_id: int) -> dict:
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    data_key, ver_key = _keys(product_id)
    try:
        cached, version = _client.mget(data_key, ver_key)
        if cached is not None:
            return json.loads(cached)
        version = version or "0"
    except valkey.ValkeyError:
        log.warning("Valkey unavailable, reading product %s from inventory", product_id, exc_info=True)
        return _fetch(product_id)

    product = _fetch(product_id)
    try:
        _SET_IF_VERSION(keys=[data_key, ver_key], args=[version, json.dumps(product), CACHE_TTL])
    except valkey.ValkeyError:
        log.warning("Could not cache product %s", product_id, exc_info=True)
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate after the inventory change is committed. Bumping the version
    # stops in-flight readers (that fetched the old price) from re-caching it.
    data_key, ver_key = _keys(product_id)
    pipe = _client.pipeline(transaction=True)
    pipe.incr(ver_key)
    pipe.delete(data_key)
    pipe.execute()
