#!/usr/bin/env python3
"""Print ids of the 5 articles most similar (cosine) to the query embedding on stdin.
The KNN search runs inside Valkey (valkey-search index 'articles_idx', FLAT/COSINE)."""
import json, struct, sys
import valkey

K = 5
vec = json.load(sys.stdin)
if len(vec) != 64:
    sys.exit(f"expected 64 floats, got {len(vec)}")
r = valkey.Valkey(host='127.0.0.1', port=6379)
res = r.execute_command(
    'FT.SEARCH', 'articles_idx', f'*=>[KNN {K} @embedding $vec AS dist]',
    'PARAMS', '2', 'vec', struct.pack('<64f', *map(float, vec)),
    'RETURN', '2', 'id', 'dist', 'LIMIT', '0', str(K), 'DIALECT', '2')
hits = []
for i in range(1, len(res), 2):
    f = res[i + 1]
    d = {f[j].decode(): f[j + 1].decode() for j in range(0, len(f), 2)}
    hits.append((float(d['dist']), d['id']))
hits.sort()  # order by Valkey-computed distance (ascending = most similar first)
for _, i in hits[:K]:
    print(i)
