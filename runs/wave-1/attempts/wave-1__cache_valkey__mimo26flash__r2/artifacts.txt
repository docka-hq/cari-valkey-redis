"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in the local Valkey server (127.0.0.1:6379) for 5 minutes
so every storefront process shares the cache instead of hitting the inventory
service.  update_price() refreshes/invalidates the cached entry, so the next
get_product() call (in any process) returns the new price.
"""
import json
import os
import urllib.request

import valkey as valkey_lib

INVENTORY_URL = "http://127.0.0.1:8000"

VALKEY_HOST = os.environ.get("VALKEY_HOST", "127.0.0.1")
VALKEY_PORT = int(os.environ.get("VALKEY_PORT", "6379"))
CACHE_TTL_SECONDS = 300  # 5 minutes
KEY_PREFIX = "catalog:product:"

# One shared client per process (valkey clients are thread-safe / pooled).
_valkey_client = None


def _get_cache():
    """Return a connected Valkey client, or None if Valkey is unreachable."""
    global _valkey_client
    if _valkey_client is None:
        try:
            client = valkey_lib.Valkey(
                host=VALKEY_HOST, port=VALKEY_PORT, decode_responses=True
            )
            client.ping()
            _valkey_client = client
        except Exception:
            return None
    return _valkey_client


def _cache_key(product_id: int) -> str:
    return f"{KEY_PREFIX}{int(product_id)}"


def get_product(product_id: int) -> dict:
    cache = _get_cache()
    if cache is not None:
        try:
            cached = cache.get(_cache_key(product_id))
        except Exception:
            cached = None
        if cached is not None:
            return json.loads(cached)

    with urllib.request.urlopen(
        f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10
    ) as resp:
        product = json.load(resp)

    if cache is not None:
        try:
            cache.set(_cache_key(product_id), json.dumps(product), ex=CACHE_TTL_SECONDS)
        except Exception:
            pass
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate the shared cache so the next get_product() in *any* process
    # re-fetches the fresh price from the inventory service.
    cache = _get_cache()
    if cache is not None:
        try:
            cache.delete(_cache_key(product_id))
        except Exception:
            pass
