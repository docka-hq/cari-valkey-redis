"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service
"""
import json
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"
CACHE_TTL_SECONDS = 300
_cache = valkey.Valkey(host="127.0.0.1", port=6379, decode_responses=True)

# A read that started before a price update must not refill the cache after
# that update invalidates it. The generation is shared across all processes.
_CACHE_IF_CURRENT = """
if (redis.call('GET', KEYS[2]) or '0') ~= ARGV[1] then
    return 0
end
redis.call('SET', KEYS[1], ARGV[2], 'EX', ARGV[3])
return 1
"""


def _keys(product_id: int) -> tuple[str, str]:
    prefix = f"storefront:catalog:product:{product_id}"
    return prefix, f"{prefix}:generation"


def get_product(product_id: int) -> dict:
    product_id = int(product_id)
    key, generation_key = _keys(product_id)
    while True:
        cached = _cache.get(key)
        if cached is not None:
            return json.loads(cached)

        generation = _cache.get(generation_key) or "0"
        with urllib.request.urlopen(f"{INVENTORY_URL}/products/{product_id}", timeout=10) as resp:
            product = json.load(resp)

        if _cache.eval(
            _CACHE_IF_CURRENT, 2, key, generation_key,
            generation, json.dumps(product), CACHE_TTL_SECONDS,
        ):
            return product
        # The price changed while the inventory read was in flight; retry.


def update_price(product_id: int, price: float) -> None:
    product_id = int(product_id)
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{product_id}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    key, generation_key = _keys(product_id)
    # One atomic operation prevents an in-flight read from restoring old data.
    with _cache.pipeline(transaction=True) as pipe:
        pipe.incr(generation_key)
        pipe.delete(key)
        pipe.execute()
