#!/usr/bin/env python3
"""Semantic search over knowledge-base articles using Redis vector search.

Usage: python3 /app/search.py < query.json
query.json contains one query embedding (a JSON list of 64 floats).
Prints the ids of the 5 most similar articles (by cosine similarity),
one per line, most similar first.

All similarity computation happens inside Redis: a KNN query against the
FLAT vector index `articles_idx` (COSINE distance, FLOAT32, DIM 64) built
over the `article:*` hashes.
"""
import json
import struct
import sys

import redis
from redis.commands.search.query import Query

REDIS_HOST = "127.0.0.1"
REDIS_PORT = 6379
INDEX = "articles_idx"
KEY_PREFIX = "article:"
K = 5
DIM = 64


def main() -> None:
    payload = json.load(sys.stdin)
    if isinstance(payload, dict):  # tolerate {"embedding": [...]} as well
        payload = payload["embedding"]
    if len(payload) != DIM:
        sys.exit(f"query embedding must have {DIM} components, got {len(payload)}")

    # Pack the query vector as little-endian FLOAT32, matching the index TYPE.
    vec_blob = struct.pack(f"<{DIM}f", *payload)

    r = redis.Redis(host=REDIS_HOST, port=REDIS_PORT)

    # KNN search inside Redis; sort by ascending cosine distance
    # (= descending cosine similarity), most similar first.
    query = (
        Query(f"*=>[KNN {K} @emb $vec AS score]")
        .sort_by("score")
        .dialect(2)
    )
    result = r.ft(INDEX).search(query, query_params={"vec": vec_blob})

    for doc in result.docs:
        key = doc.id  # e.g. "article:kb-0042"
        print(key[len(KEY_PREFIX):] if key.startswith(KEY_PREFIX) else key)


if __name__ == "__main__":
    main()
