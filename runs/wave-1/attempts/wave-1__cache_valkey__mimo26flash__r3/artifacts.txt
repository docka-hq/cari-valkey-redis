"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products read through get_product() are cached in the local Valkey server
(127.0.0.1:6379) so that every storefront process shares one cache instead of
hitting the inventory service.  Cache entries expire after 5 minutes, and
update_price() invalidates the entry so the next read gets the fresh price.
"""
import json
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"

VALKEY_HOST = "127.0.0.1"
VALKEY_PORT = 6379
CACHE_TTL_SECONDS = 300  # 5 minutes
KEY_PREFIX = "catalog:product:"


def _cache_key(product_id: int) -> str:
    return f"{KEY_PREFIX}{int(product_id)}"


def _get_valkey() -> valkey.Valkey:
    return valkey.Valkey(host=VALKEY_HOST, port=VALKEY_PORT, decode_responses=True)


def get_product(product_id: int) -> dict:
    product_id = int(product_id)
    client = _get_valkey()

    cached = client.get(_cache_key(product_id))
    if cached is not None:
        return json.loads(cached)

    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{product_id}", timeout=10) as resp:
        product = json.load(resp)

    client.set(_cache_key(product_id), json.dumps(product), ex=CACHE_TTL_SECONDS)
    return product


def update_price(product_id: int, price: float) -> None:
    product_id = int(product_id)
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{product_id}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate the shared cache so the next get_product() (in any process)
    # returns the new price.
    _get_valkey().delete(_cache_key(product_id))
