"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products read through get_product() are cached in the shared Redis server
(127.0.0.1:6379) for 5 minutes, so every storefront process reuses the same
cached copy instead of calling the slow inventory service again.  update_price()
invalidates the cached copy so the next get_product() call -- in any process --
sees the new price.
"""
import json
import os
import threading
import urllib.request

import redis

INVENTORY_URL = os.environ.get("INVENTORY_URL", "http://127.0.0.1:8000")
REDIS_URL = os.environ.get("REDIS_URL", "redis://127.0.0.1:6379/0")

CACHE_TTL_SECONDS = 300  # 5 minutes
CACHE_KEY_PREFIX = "catalog:product:"

_client = None
_client_lock = threading.Lock()


def _cache() -> redis.Redis:
    """Lazily create one thread-safe Redis client per process."""
    global _client
    if _client is None:
        with _client_lock:
            if _client is None:
                _client = redis.Redis.from_url(REDIS_URL, decode_responses=True)
    return _client


def _cache_key(product_id: int) -> str:
    return f"{CACHE_KEY_PREFIX}{int(product_id)}"


def get_product(product_id: int) -> dict:
    key = _cache_key(product_id)

    # Serve recently-read products straight from the shared Redis cache.
    try:
        cached = _cache().get(key)
    except redis.RedisError:
        cached = None  # cache unavailable -> fall through to the inventory service
    if cached is not None:
        return json.loads(cached)

    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        product = json.load(resp)

    try:
        _cache().setex(key, CACHE_TTL_SECONDS, json.dumps(product))
    except redis.RedisError:
        pass  # caching is best-effort; still return the fresh product
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        updated = json.load(resp)

    # Invalidate/refresh the shared cache so every process sees the new price
    # on its next get_product() call.
    key = _cache_key(product_id)
    try:
        _cache().setex(key, CACHE_TTL_SECONDS, json.dumps(updated))
    except redis.RedisError:
        try:
            _cache().delete(key)
        except redis.RedisError:
            pass
