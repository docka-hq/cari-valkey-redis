#!/usr/bin/env python3
"""Semantic search over knowledge-base articles stored in Redis.

Usage:  python3 /app/search.py < query.json
where query.json is a JSON list of 64 floats (the query embedding).

Runs a k-nearest-neighbour (cosine) similarity search inside Redis via
RediSearch (FT.SEARCH with KNN over a FLAT vector index).  Reads nothing
from disk and computes no similarities locally.
"""
import json
import struct
import sys

import redis

INDEX = "idx:articles"
TOP_K = 5


def main() -> None:
    query = json.load(sys.stdin)
    if not isinstance(query, list) or len(query) != 64:
        sys.exit("expected a JSON list of 64 floats on stdin")
    # Raw little-endian float32 bytes: the index stores vectors as FLOAT32.
    vec = struct.pack("<64f", *(float(x) for x in query))

    r = redis.Redis(host="127.0.0.1", port=6379, decode_responses=False)
    res = r.execute_command(
        "FT.SEARCH",
        INDEX,
        f"*=>[KNN {TOP_K} @embedding $vec AS distance]",
        "PARAMS", "2", "vec", vec,
        "SORTBY", "distance", "ASC",
        "RETURN", "1", "id",
        "DIALECT", "2",
    )

    results = []
    if isinstance(res, dict):
        results = res.get(b"results", res.get("results", []))
    else:  # RESP2 list format: [total, key, fields, key, fields, ...]
        results = [
            dict(zip(res[i + 1][::2], res[i + 1][1::2]))
            for i in range(1, len(res), 2)
        ]

    for hit in results:
        attrs = hit.get(b"extra_attributes", hit.get("extra_attributes", hit))
        art_id = attrs.get(b"id", attrs.get("id"))
        if art_id is None:  # fall back to the key name "art:<id>"
            key = hit.get(b"id", hit.get("id", b"")).decode()
            art_id = key.split(":", 1)[-1]
        print(art_id.decode() if isinstance(art_id, bytes) else art_id)


if __name__ == "__main__":
    main()
