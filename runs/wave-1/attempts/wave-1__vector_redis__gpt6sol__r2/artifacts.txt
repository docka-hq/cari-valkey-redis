#!/usr/bin/env python3
"""Print the five nearest knowledge-base IDs using Redis cosine vector search."""
import json
import math
import struct
import sys

import redis


def main():
    embedding = json.load(sys.stdin)
    if (not isinstance(embedding, list) or len(embedding) != 64
            or any(isinstance(x, bool) or not isinstance(x, (float, int))
                   or not math.isfinite(x) for x in embedding)):
        raise ValueError('Expected a JSON list of 64 finite numbers')

    vector = struct.pack('<64d', *embedding)
    client = redis.Redis(host='127.0.0.1', port=6379)
    result = client.execute_command(
        'FT.SEARCH', 'idx:kb:articles',
        '*=>[KNN 5 @embedding $query AS distance]',
        'PARAMS', 2, 'query', vector,
        'SORTBY', 'distance', 'ASC', 'RETURN', 1, 'id',
        'LIMIT', 0, 5, 'DIALECT', 2,
    )
    # Redis 8 RESP2 returns a map containing the ordered result documents.
    for document in result[b'results']:
        print(document[b'extra_attributes'][b'id'].decode('utf-8'))


if __name__ == '__main__':
    main()
