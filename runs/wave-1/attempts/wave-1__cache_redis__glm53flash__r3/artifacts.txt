"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Product reads go through the shared Redis cache (127.0.0.1:6379, db 0). A
product that was read recently is served from Redis, so every storefront
process shares the same cache and the inventory service is not called again.
Cache entries expire after CACHE_TTL_SECONDS (5 minutes). update_price()
deletes the cache entry, so the next get_product() for that product -- in any
process -- fetches and returns the new price.

If Redis is unavailable, reads and writes fall back to the inventory service.
"""
import json
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"
REDIS_URL = "redis://127.0.0.1:6379/0"
CACHE_TTL_SECONDS = 5 * 60  # cached products expire after 5 minutes

# One client per process; redis-py's connection pool behind it is thread-safe.
_redis = redis.Redis.from_url(
    REDIS_URL,
    decode_responses=True,
    socket_connect_timeout=1,
    socket_timeout=1,  # never let a slow Redis stall a storefront read
)


def _cache_key(product_id: int) -> str:
    return f"product:{int(product_id)}"


def get_product(product_id: int) -> dict:
    key = _cache_key(product_id)

    # Serve the product from the shared Redis cache when it was read recently.
    try:
        cached = _redis.get(key)
    except redis.RedisError:
        cached = None
    if cached is not None:
        try:
            return json.loads(cached)
        except ValueError:
            pass  # corrupt entry: treat it as a cache miss

    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        product = json.load(resp)

    # Fill the cache for the other storefront processes; expires after 5 minutes.
    try:
        _redis.set(key, json.dumps(product), ex=CACHE_TTL_SECONDS)
    except redis.RedisError:
        pass  # return the product even if caching failed
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # The cache now holds a stale price: drop it so the next get_product() in
    # any process fetches the new price from the inventory service.
    try:
        _redis.delete(_cache_key(product_id))
    except redis.RedisError:
        pass
