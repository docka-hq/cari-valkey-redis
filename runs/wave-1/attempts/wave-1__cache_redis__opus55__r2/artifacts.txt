"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in Redis (shared by all storefront processes) for
CACHE_TTL_SECONDS.  update_price() invalidates the cached entry, and a per-product
version key guarded by WATCH/MULTI prevents a reader that fetched the old data
before the update from writing it back into the cache afterwards.
"""
import json
import logging
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"

REDIS_URL = "redis://127.0.0.1:6379/0"
CACHE_TTL_SECONDS = 300
_KEY_PREFIX = "catalog:product:"

log = logging.getLogger(__name__)

# redis-py's connection pool is thread-safe and fork-aware (it resets itself
# in a child process), so one module-level client is fine for every process.
_redis = redis.Redis.from_url(
    REDIS_URL, socket_timeout=1, socket_connect_timeout=1, decode_responses=True
)


def _cache_key(product_id: int) -> str:
    return f"{_KEY_PREFIX}{product_id}"


def _version_key(product_id: int) -> str:
    return f"{_KEY_PREFIX}{product_id}:version"


def _fetch_product(product_id: int) -> dict:
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{product_id}", timeout=10) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    product_id = int(product_id)
    key = _cache_key(product_id)

    try:
        cached = _redis.get(key)
    except redis.RedisError as exc:
        log.warning("Redis unavailable, reading product %s from inventory: %s", product_id, exc)
        return _fetch_product(product_id)
    if cached is not None:
        return json.loads(cached)

    # Cache miss: fetch from inventory and store it, but only if no
    # update_price() happened for this product while we were fetching.
    try:
        with _redis.pipeline(transaction=True) as pipe:
            pipe.watch(_version_key(product_id))
            product = _fetch_product(product_id)
            pipe.multi()
            pipe.set(key, json.dumps(product), ex=CACHE_TTL_SECONDS)
            pipe.execute()
    except redis.WatchError:
        pass  # price changed concurrently; don't cache possibly stale data
    except redis.RedisError as exc:
        log.warning("Could not cache product %s: %s", product_id, exc)
        if "product" not in locals():
            return _fetch_product(product_id)
    return product


def update_price(product_id: int, price: float) -> None:
    product_id = int(product_id)
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{product_id}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate the cached product and bump its version so any in-flight
    # cache fill that read the old price is discarded (WATCH fails).
    with _redis.pipeline(transaction=True) as pipe:
        pipe.incr(_version_key(product_id))
        pipe.delete(_cache_key(product_id))
        pipe.execute()
