import json, sys
import numpy as np
import redis

q = np.array(json.load(sys.stdin), dtype=np.float32)
if q.shape != (64,):
    sys.exit("query must be a list of 64 floats")
r = redis.Redis(host="127.0.0.1", port=6379, protocol=2)
res = r.execute_command(
    "FT.SEARCH", "idx:articles", "*=>[KNN 5 @embedding $vec AS score]",
    "PARAMS", "2", "vec", q.tobytes(),
    "SORTBY", "score", "ASC", "RETURN", "1", "id", "LIMIT", "0", "5", "DIALECT", "2",
)
for i in range(2, len(res), 2):
    f = res[i]
    d = {f[j]: f[j + 1] for j in range(0, len(f), 2)}
    print(d[b"id"].decode())
