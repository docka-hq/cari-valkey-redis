#!/usr/bin/env python3
"""Semantic search over the knowledge-base article vectors stored in Redis.

Usage:
    python3 /app/search.py < query.json

query.json must contain a single query embedding: a JSON list of floats
(e.g. [0.01, -0.2, ...]). The 5 most similar article ids are printed to
stdout, one per line, most similar first.

The similarity search runs entirely inside Redis (VSIM over the vector set
`articles:vset`): this script never reads /data/articles.jsonl and never
computes similarities itself.
"""
import json
import sys

import redis

VSET_KEY = "articles:vset"
TOP_K = 5


def read_query() -> list[float]:
    raw = sys.stdin.read().strip()
    if not raw:
        raise SystemExit("error: empty query on stdin")
    try:
        query = json.loads(raw)
    except json.JSONDecodeError as exc:
        raise SystemExit(f"error: invalid JSON query: {exc}")

    # Tolerate an object form such as {"embedding": [...]} / {"vector": [...]}.
    if isinstance(query, dict):
        for field in ("embedding", "vector", "query"):
            if field in query:
                query = query[field]
                break
    if not isinstance(query, (list, tuple)):
        raise SystemExit("error: query must be a JSON list of floats")
    try:
        return [float(x) for x in query]
    except (TypeError, ValueError):
        raise SystemExit("error: query must contain only numbers")


def main() -> int:
    query = read_query()

    r = redis.Redis(host="127.0.0.1", port=6379, decode_responses=True)
    dim = int(r.execute_command("VDIM", VSET_KEY))
    if len(query) != dim:
        raise SystemExit(f"error: query has {len(query)} dims, index expects {dim}")

    # VSIM ... VALUES <dim> <v1..vN> COUNT 5 WITHSCORES TRUTH
    # TRUTH forces an exact (non-approximate) search over the vector set, so the
    # ranking is exact for this collection size.
    result = r.execute_command(
        "VSIM", VSET_KEY, "VALUES", str(dim),
        *[repr(v) for v in query],
        "COUNT", str(TOP_K), "WITHSCORES", "TRUTH",
    )

    # redis-py may return the reply as a flat [id, score, id, score, ...] list
    # or as a mapping id -> score, depending on RESP version.
    if isinstance(result, dict):
        scored = [(str(k), float(v)) for k, v in result.items()]
    elif result and isinstance(result[0], dict):
        scored = [(str(k), float(v)) for k, v in result[0].items()]
    else:
        scored = [(str(result[i]), float(result[i + 1])) for i in range(0, len(result), 2)]

    scored.sort(key=lambda kv: kv[1], reverse=True)
    out = [article_id for article_id, _ in scored[:TOP_K]]
    sys.stdout.write("\n".join(out) + ("\n" if out else ""))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
