"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service
"""
import json
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"
CACHE_TTL_SECONDS = 300
_cache = redis.Redis(host="127.0.0.1", port=6379, db=0)


def _product_key(product_id: int) -> str:
    return f"catalog:product:{product_id}"


def _product_lock(product_id: int):
    # Serialize misses and writes across storefront processes. In particular,
    # a read started before a price update cannot refill the cache afterward.
    return _cache.lock(f"catalog:product-lock:{product_id}", timeout=30)


def get_product(product_id: int) -> dict:
    product_id = int(product_id)
    key = _product_key(product_id)
    cached = _cache.get(key)
    if cached is not None:
        return json.loads(cached)

    with _product_lock(product_id):
        # Another process may have populated the cache while we waited.
        cached = _cache.get(key)
        if cached is not None:
            return json.loads(cached)
        with urllib.request.urlopen(f"{INVENTORY_URL}/products/{product_id}", timeout=10) as resp:
            product = json.load(resp)
        _cache.setex(key, CACHE_TTL_SECONDS, json.dumps(product))
        return product


def update_price(product_id: int, price: float) -> None:
    product_id = int(product_id)
    with _product_lock(product_id):
        body = json.dumps({"price": price}).encode()
        req = urllib.request.Request(
            f"{INVENTORY_URL}/products/{product_id}",
            data=body,
            method="PUT",
            headers={"Content-Type": "application/json"},
        )
        with urllib.request.urlopen(req, timeout=10) as resp:
            resp.read()
        # Invalidate only after a successful update. The next read will fetch
        # the new price; the lock prevents an older in-flight miss from winning.
        _cache.delete(_product_key(product_id))
