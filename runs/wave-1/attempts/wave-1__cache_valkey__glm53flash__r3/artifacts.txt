"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in Valkey (shared by every storefront process) for
CACHE_TTL_SECONDS, so repeated get_product() calls do not hit the
inventory service.  update_price() invalidates the cached entry, so the
next read - in any process - sees the new price.
"""
import json
import threading
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"

VALKEY_HOST = "127.0.0.1"
VALKEY_PORT = 6379
CACHE_TTL_SECONDS = 300  # cached products expire after 5 minutes

_KEY_PREFIX = "catalog:product"

_client = None
_client_lock = threading.Lock()


def _cache_key(product_id: int) -> str:
    return f"{_KEY_PREFIX}:{int(product_id)}"


def _get_client() -> valkey.Valkey:
    """Return the shared Valkey client, creating it once per process."""
    global _client
    if _client is None:
        with _client_lock:
            if _client is None:
                _client = valkey.Valkey(
                    host=VALKEY_HOST,
                    port=VALKEY_PORT,
                    socket_timeout=1,
                    socket_connect_timeout=1,
                )
    return _client


def _cache_get_json(key: str):
    """Read JSON from Valkey; None on miss or if Valkey is unreachable."""
    try:
        raw = _get_client().get(key)
    except valkey.exceptions.ValkeyError:
        return None  # cache is best-effort; fall through to the source
    if raw is None:
        return None
    try:
        return json.loads(raw)
    except (ValueError, TypeError):
        return None


def _cache_set_json(key: str, value: dict) -> None:
    try:
        _get_client().set(key, json.dumps(value), ex=CACHE_TTL_SECONDS)
    except valkey.exceptions.ValkeyError:
        pass  # never let the cache take the store down


def _cache_delete(key: str) -> None:
    try:
        _get_client().delete(key)
    except valkey.exceptions.ValkeyError:
        pass  # the entry will still expire after the TTL


def get_product(product_id: int) -> dict:
    key = _cache_key(product_id)
    cached = _cache_get_json(key)
    if cached is not None:
        return cached
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        product = json.load(resp)
    _cache_set_json(key, product)
    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()
    _cache_delete(_cache_key(product_id))
