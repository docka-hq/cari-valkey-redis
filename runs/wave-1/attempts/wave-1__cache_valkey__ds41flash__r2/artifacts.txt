"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in Valkey so that every storefront process shares the same
cache and repeated reads do not hit the inventory service.  Cached entries
expire after CACHE_TTL seconds.  update_price() invalidates the cache entry for
the product it changes, so the next get_product() call (in any process) reads
the new price from the inventory service.
"""
import json
import urllib.request

import valkey

INVENTORY_URL = "http://127.0.0.1:8000"

VALKEY_HOST = "127.0.0.1"
VALKEY_PORT = 6379
CACHE_TTL = 300  # seconds (5 minutes)
KEY_PREFIX = "catalog:product:"

# One shared, thread-safe connection pool per process.
_client = valkey.Valkey(
    host=VALKEY_HOST,
    port=VALKEY_PORT,
    decode_responses=True,
    socket_timeout=2,
)


def _cache_key(product_id: int) -> str:
    return f"{KEY_PREFIX}{int(product_id)}"


def _fetch_from_inventory(product_id: int) -> dict:
    with urllib.request.urlopen(
        f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10
    ) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    key = _cache_key(product_id)

    try:
        cached = _client.get(key)
    except valkey.ValkeyError:
        cached = None

    if cached is not None:
        return json.loads(cached)

    product = _fetch_from_inventory(product_id)

    try:
        _client.set(key, json.dumps(product), ex=CACHE_TTL)
    except valkey.ValkeyError:
        pass

    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate the shared cache entry so every process sees the new price.
    try:
        _client.delete(_cache_key(product_id))
    except valkey.ValkeyError:
        pass
