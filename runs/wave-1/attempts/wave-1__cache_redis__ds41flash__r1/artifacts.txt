"""
import json
import time
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"

# --- Redis cache configuration --------------------------------------------
REDIS_URL = "redis://127.0.0.1:6379/0"
CACHE_TTL = 300  # seconds: a cached product expires after 5 minutes

# Version markers live longer than the entries they guard, so a version
# counter can never be reset (and then reused) while an entry written under
# the previous version could still be cached.
_VERSION_TTL = CACHE_TTL * 12

# Single-flight: while one process fetches a cold key the others wait (briefly)
# for the shared cache instead of all hammering the inventory service.
_LOCK_TTL = 10    # seconds the fetch lock may live before it expires
_WAIT = 1.0       # seconds a waiter waits for the leader before fetching itself
_POLL = 0.02      # seconds between waiter cache checks
_RETRIES = 4      # times to restart a read that was invalidated mid-flight

# A single module-level client shared by all threads in the process.  redis-py's
# connection pool is fork-safe (it notices a changed PID and reconnects), so
# importing this module in a forked worker just works.
_redis = redis.Redis.from_url(
    REDIS_URL, decode_responses=True, socket_timeout=2, socket_connect_timeout=2
)


def _cache_key(product_id: int, version: int) -> str:
    return f"catalog:product:{product_id}:v{version}"


def _version_key(product_id: int) -> str:
    return f"catalog:product:{product_id}:version"


def _lock_key(product_id: int, version: int) -> str:
    return f"catalog:product:{product_id}:v{version}:lock"


def _get_version(product_id: int) -> int:
    """Current cache generation for a product (0 when it was never invalidated)."""
    value = _redis.get(_version_key(product_id))
    return int(value) if value is not None else 0


def _bump_version(product_id: int) -> int:
    """Invalidate every cached entry for a product, returning the new version."""
    pipe = _redis.pipeline()
    pipe.incr(_version_key(product_id))
    pipe.expire(_version_key(product_id), _VERSION_TTL)
    version, _ = pipe.execute()
    return int(version)


def _fetch_product(product_id: int) -> dict:
    with urllib.request.urlopen(f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    product_id = int(product_id)

    for attempt in range(_RETRIES):
        # Read the current version together with the entry cached under it.  A
        # cache failure must never take the storefront down, so fall back to
        # the inventory service if Redis is unavailable.
        try:
            version = _get_version(product_id)
            key = _cache_key(product_id, version)
            cached = _redis.get(key)
        except redis.RedisError:
            return _fetch_product(product_id)

        if cached is not None:
            return json.loads(cached)

        # Cache miss.  Let a single process fetch while the others briefly wait
        # for it, so a cold (or just-invalidated) key does not stampede the
        # inventory service when many processes ask for it at once.
        lock_key = _lock_key(product_id, version)
        try:
            is_leader = _redis.set(lock_key, "1", nx=True, ex=_LOCK_TTL)
        except redis.RedisError:
            return _fetch_product(product_id)

        if not is_leader:
            deadline = time.monotonic() + _WAIT
            while time.monotonic() < deadline:
                time.sleep(_POLL)
                try:
                    cached = _redis.get(key)
                    if cached is not None:
                        return json.loads(cached)
                    if _get_version(product_id) != version:
                        break  # invalidated while we waited: restart the read
                    if not _redis.exists(lock_key):
                        break  # leader gave up: fetch it ourselves
                except redis.RedisError:
                    break
            else:
                # Waited the full time and the leader is still not done.
                return _fetch_product(product_id)
            continue  # retry with the (possibly new) version

        # Leader: fetch once and populate the shared cache.
        try:
            product = _fetch_product(product_id)
        except BaseException:
            try:
                _redis.delete(lock_key)  # let waiters retry promptly
            except redis.RedisError:
                pass
            raise
        try:
            _redis.set(key, json.dumps(product), ex=CACHE_TTL)
        except redis.RedisError:
            pass
        finally:
            try:
                _redis.delete(lock_key)
            except redis.RedisError:
                pass
        return product

    return _fetch_product(product_id)


def update_price(product_id: int, price: float) -> None:
    product_id = int(product_id)
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{product_id}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate the cache *after* the inventory write has completed, so no
    # process keeps serving the old price.  Bumping the version makes every
    # entry cached under the previous version unreachable; that also prevents a
    # reader that is concurrently fetching the old value from resurrecting it,
    # because it would store it under the now-obsolete version.
    try:
        _bump_version(product_id)
    except redis.RedisError:
        # Best effort: at least drop whatever is cached right now.
        try:
            _redis.delete(_cache_key(product_id, _get_version(product_id)))
        except redis.RedisError:
            pass
