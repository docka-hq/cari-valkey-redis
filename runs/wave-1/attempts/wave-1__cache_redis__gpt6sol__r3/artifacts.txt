"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service
"""
import json
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"
_CACHE_TTL = 300
_cache = redis.Redis(host="127.0.0.1", port=6379, decode_responses=True)

# Only populate the cache if no price update completed while the inventory
# request was in flight. Checking and setting must be one atomic operation.
_CACHE_IF_CURRENT = """
if (redis.call('GET', KEYS[2]) or '0') ~= ARGV[1] then
    return 0
end
redis.call('SET', KEYS[1], ARGV[2], 'EX', ARGV[3])
return 1
"""
_INVALIDATE = """
redis.call('INCR', KEYS[2])
redis.call('DEL', KEYS[1])
"""


def _keys(product_id: int) -> tuple[str, str]:
    key = f"storefront:catalog:product:{int(product_id)}"
    return key, f"{key}:version"


def get_product(product_id: int) -> dict:
    product_id = int(product_id)
    key, version_key = _keys(product_id)
    while True:
        cached = _cache.get(key)
        if cached is not None:
            return json.loads(cached)

        version = _cache.get(version_key) or "0"
        with urllib.request.urlopen(f"{INVENTORY_URL}/products/{product_id}", timeout=10) as resp:
            product = json.load(resp)

        if _cache.eval(
            _CACHE_IF_CURRENT, 2, key, version_key,
            version, json.dumps(product), _CACHE_TTL,
        ):
            return product
        # An update won the race against this read: do not return old data.


def update_price(product_id: int, price: float) -> None:
    product_id = int(product_id)
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{product_id}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # The version is intentionally persistent: an in-flight read must never
    # mistake an expired/recreated version for the one it observed earlier.
    key, version_key = _keys(product_id)
    _cache.eval(_INVALIDATE, 2, key, version_key)
