"""Product catalog used by the storefront.

get_product(product_id) -> dict    {"id": int, "name": str, "price": float, "currency": str}
update_price(product_id, price)    changes the price in the inventory service

Products are cached in a shared Redis instance so that every storefront
process serves recently-read products without hitting the inventory service
again.  Cached entries expire after CACHE_TTL seconds (5 minutes).  Whenever a
price is updated the cache entry is invalidated, so the next get_product()
call for that product - in any process - returns the new price.
"""
import json
import urllib.request

import redis

INVENTORY_URL = "http://127.0.0.1:8000"

# Shared cache, reachable by every storefront process.
REDIS_HOST = "127.0.0.1"
REDIS_PORT = 6379
CACHE_TTL = 300  # seconds (5 minutes)
KEY_PREFIX = "catalog:product:"
VERSION_PREFIX = "catalog:version:"

_redis = redis.Redis(
    host=REDIS_HOST,
    port=REDIS_PORT,
    decode_responses=True,
    socket_timeout=2.0,
    socket_connect_timeout=2.0,
    health_check_interval=30,
)

# Atomically store a freshly fetched product, but only if the product has not
# been updated (version changed) in the meantime.  This prevents a slow
# inventory read from re-populating the cache with a stale value right after
# update_price() invalidated it.
_STORE_IF_UNCHANGED = """
local current = redis.call('GET', KEYS[2])
if not current then current = '0' end
if current == ARGV[1] then
    redis.call('SET', KEYS[1], ARGV[2], 'EX', ARGV[3])
    return 1
end
return 0
"""
_store_if_unchanged = _redis.register_script(_STORE_IF_UNCHANGED)


def _product_key(product_id: int) -> str:
    return f"{KEY_PREFIX}{int(product_id)}"


def _version_key(product_id: int) -> str:
    return f"{VERSION_PREFIX}{int(product_id)}"


def _fetch_from_inventory(product_id: int) -> dict:
    with urllib.request.urlopen(
        f"{INVENTORY_URL}/products/{int(product_id)}", timeout=10
    ) as resp:
        return json.load(resp)


def get_product(product_id: int) -> dict:
    key = _product_key(product_id)
    version_key = _version_key(product_id)

    # 1. Try the shared cache first.
    try:
        version = _redis.get(version_key) or "0"
        cached = _redis.get(key)
        if cached is not None:
            return json.loads(cached)
    except redis.RedisError:
        # Cache unavailable: degrade gracefully to the inventory service.
        version = None
        pass

    # 2. Cache miss: read the product from the inventory service.
    product = _fetch_from_inventory(product_id)

    # 3. Populate the cache for every process, with a 5 minute TTL.
    if version is not None:
        try:
            _store_if_unchanged(
                keys=[key, version_key],
                args=[version, json.dumps(product), CACHE_TTL],
            )
        except redis.RedisError:
            pass

    return product


def update_price(product_id: int, price: float) -> None:
    body = json.dumps({"price": price}).encode()
    req = urllib.request.Request(
        f"{INVENTORY_URL}/products/{int(product_id)}",
        data=body,
        method="PUT",
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as resp:
        resp.read()

    # Invalidate the shared cache entry so the next read (in any process)
    # fetches the new price.  Bumping the version also blocks an in-flight
    # get_product() from caching the pre-update value.
    try:
        _redis.delete(_product_key(product_id))
        _redis.incr(_version_key(product_id))
    except redis.RedisError:
        pass
