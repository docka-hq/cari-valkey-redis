#!/usr/bin/env python3
"""Semantic search over knowledge-base articles stored in Valkey.

Usage:
    python3 /app/search.py < query.json

`query.json` contains a single query embedding: a JSON list of 64 floats.

The 5 most similar articles (by cosine similarity) are printed, one `id`
per line, most similar first.

All similarity computation is performed *inside Valkey* by the Valkey
Search module (FT.SEARCH ... KNN).  This script never reads the source
articles file and never computes similarities itself -- it only sends the
query vector to Valkey and prints the ids that Valkey returns, already
ranked by the engine.
"""

import json
import struct
import sys

import valkey

HOST = "127.0.0.1"
PORT = 6379
INDEX = "articles_idx"
DIM = 64
K = 5


def main() -> int:
    raw = sys.stdin.read().strip()
    if not raw:
        print("error: empty query", file=sys.stderr)
        return 1

    try:
        query = json.loads(raw)
    except json.JSONDecodeError as exc:
        print(f"error: invalid JSON query: {exc}", file=sys.stderr)
        return 1

    if not isinstance(query, list) or len(query) != DIM:
        print(f"error: expected a JSON list of {DIM} floats", file=sys.stderr)
        return 1

    # Pack the query vector as little-endian float32 bytes, exactly the
    # representation stored in the index (FLOAT32).
    blob = struct.pack(f"<{DIM}f", *[float(x) for x in query])

    client = valkey.Valkey(host=HOST, port=PORT, decode_responses=True)

    # KNN vector search executed by Valkey Search.  Valkey returns the
    # matches ordered by cosine distance, ascending (i.e. most similar
    # first).  We ask it to return only the distance it computed.
    res = client.execute_command(
        "FT.SEARCH", INDEX,
        "(*)=>[KNN {} @embedding $vec AS dist]".format(K),
        "PARAMS", 2, "vec", blob,
        "RETURN", 1, "dist",
        "DIALECT", 2,
    )

    # Response shape: [total, key1, [field, value, ...], key2, [...], ...]
    ranked = []
    for i in range(1, len(res), 2):
        key = res[i]
        fields = res[i + 1] if i + 1 < len(res) else []
        fmap = dict(zip(fields[::2], fields[1::2]))
        doc_id = key.split(":", 1)[1] if ":" in key else key
        ranked.append((float(fmap.get("dist", 0.0)), doc_id))

    # Belt-and-suspenders: order by the distances Valkey computed.
    ranked.sort(key=lambda t: t[0])

    for _dist, doc_id in ranked[:K]:
        print(doc_id)

    return 0


if __name__ == "__main__":
    sys.exit(main())
